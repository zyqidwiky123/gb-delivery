# Workflow 3.2 — Setup Minimal Testing

**Severity:** 🟢 MEDIUM
**Estimasi:** 2-3 hari
**Dependensi:** 2.1 (split index.js)

---

## Langkah

### 1. Backend Testing (Jest)

```bash
cd functions
npm install --save-dev jest
```

Buat `functions/jest.config.js`:

```javascript
module.exports = {
    testEnvironment: 'node',
    testMatch: ['**/tests/**/*.test.js']
};
```

Buat `functions/tests/`:

**tests/pricing.test.js:**

```javascript
const { calculateFee, calculateAppFee } = require('../src/pricing');

describe('calculateAppFee', () => {
    test('jarak 3km → Rp 500 (1 block)', () => {
        expect(calculateAppFee(3, 4, 500)).toBe(500);
    });
    test('jarak 5km → Rp 1.000 (2 block)', () => {
        expect(calculateAppFee(5, 4, 500)).toBe(1000);
    });
    test('jarak 0km → Rp 500 (1 block)', () => {
        expect(calculateAppFee(0, 4, 500)).toBe(500);
    });
});
```

**tests/dispatch.test.js:**

```javascript
const { calculateDistance } = require('../src/utils');

describe('calculateDistance', () => {
    test('jarak 0 (koordinat sama) → 0', () => {
        expect(calculateDistance(-7.5, 112.5, -7.5, 112.5)).toBe(0);
    });
    test('jarak Blitar-Malang ~60km', () => {
        const dist = calculateDistance(-8.0956, 112.1641, -7.9797, 112.6304);
        expect(dist).toBeGreaterThan(50);
        expect(dist).toBeLessThan(80);
    });
});
```

**tests/validation.test.js:**

```javascript
const { validateFirebaseUid, validateEmail } = require('../src/validation');

describe('validateFirebaseUid', () => {
    test('uid valid → true', () => {
        expect(validateFirebaseUid('abc123-DEF_456')).toBe(true);
    });
    test('uid kosong → false', () => {
        expect(validateFirebaseUid('')).toBe(false);
    });
    test('uid dengan simbol invalid → false', () => {
        expect(validateFirebaseUid('abc@123')).toBe(false);
    });
});
```

### 2. Frontend Testing (Vitest)

```bash
cd aro-drive-vite
npm install --save-dev vitest @testing-library/react @testing-library/jest-dom jsdom
```

Buat `aro-drive-vite/vitest.config.js`:

```javascript
import { defineConfig } from 'vitest/config';
import react from '@vitejs/plugin-react';

export default defineConfig({
    plugins: [react()],
    test: {
        environment: 'jsdom',
        globals: true
    }
});
```

Buat `aro-drive-vite/src/store/__tests__/orderStore.test.js`:

```javascript
import { describe, it, expect } from 'vitest';
import { calculateAppServiceFee } from '../../utils/pricing';

describe('calculateAppServiceFee', () => {
    it('returns correct fee for distance', () => {
        expect(calculateAppServiceFee(3, 4, 500)).toBe(500);
        expect(calculateAppServiceFee(5, 4, 500)).toBe(1000);
    });
});
```

### 3. Run & Verifikasi

```bash
cd functions && npx jest
cd aro-drive-vite && npx vitest run
```

### 4. Integrasi CI/CD

Di `deploy.yml`, tambah sebelum deploy:

```yaml
- name: Run Backend Tests
  run: cd functions && npm test

- name: Run Frontend Tests
  run: cd aro-drive-vite && npm test
```
