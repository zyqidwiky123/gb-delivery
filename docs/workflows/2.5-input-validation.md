## Langkah

### 1. Buat functions/src/validation.js

```javascript
// Helper validasi
function isNonEmptyString(val) {
    return typeof val === 'string' && val.trim().length > 0;
}

function validateFirebaseUid(uid) {
    if (!isNonEmptyString(uid)) return false;
    // Firebase UID maks 128 karakter, alphanumeric
    return uid.length <= 128 && /^[a-zA-Z0-9_-]+$/.test(uid);
}

function validateEmail(email) {
    if (!isNonEmptyString(email)) return false;
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

function validateOrderId(orderId) {
    if (!isNonEmptyString(orderId)) return false;
    return orderId.length <= 64;
}

function sanitizeString(str, maxLen = 500) {
    if (!str) return '';
    return String(str).trim().substring(0, maxLen);
}

function validatePhoneNumber(phone) {
    if (!isNonEmptyString(phone)) return false;
    return /^\+?[0-9]{10,15}$/.test(phone.replace(/\s/g, ''));
}

module.exports = {
    isNonEmptyString,
    validateFirebaseUid,
    validateEmail,
    validateOrderId,
    sanitizeString,
    validatePhoneNumber
};
```

### 2. Terapkan ke Endpoint

Contoh di `deleteUserAccount`:

```javascript
const { uid } = req.body;
if (!validateFirebaseUid(uid)) {
    res.status(400).json({ error: 'Invalid uid' });
    return;
}
```
